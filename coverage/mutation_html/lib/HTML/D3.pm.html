 1111: 
 1112:         // Draw lines for each series
 1113:         data.forEach((series, i) => {
 1114:             const line = d3.line()
 1115:                 .x(d => x(d.label))
 1116:                 .y(d => y(d.value));
 1117: 
 1118:             // Add line
 1119:             chart.append("path")
 1120:                 .datum(series.data)
 1121:                 .attr("fill", "none")
 1122:                 .attr("stroke", color(i))
 1123:                 .attr("stroke-width", 2)
 1124:                 .attr("class", \`line-\${i}\`)
 1125:                 .attr("d", line);
 1126: 
 1127:             // Add points and tooltips
 1128:             chart.selectAll(\`circle.series-\${i}\`)
 1129:                 .data(series.data)
 1130:                 .join("circle")
 1131:                 .attr("class", \`series-\${i}\`)
 1132:                 .attr("cx", d => x(d.label))
 1133:                 .attr("cy", d => y(d.value))
 1134:                 .attr("r", 4)
 1135:                 .attr("fill", color(i))
 1136:                 .on("mouseover", (event, d) => {
 1137:                     tooltip.style("opacity", 1)
 1138:                            .style("transform", "translateY(0)")
 1139:                            .html(\`Series: <b>\${series.name}</b><br>Label: <b>\${d.label}</b><br>Value: <b>\${d.value}</b>\`)
 1140:                            .style("left", (event.pageX + 10) + "px")
 1141:                            .style("top", (event.pageY - 30) + "px");
 1142:                 })
 1143:                 .on("mousemove", (event) => {
 1144:                     tooltip.style("left", (event.pageX + 10) + "px")
 1145:                            .style("top", (event.pageY - 30) + "px");
 1146:                 })
 1147:                 .on("mouseout", () => {
 1148:                     tooltip.style("opacity", 0)
 1149:                            .style("transform", "translateY(-10px)");
 1150:                 });
 1151:         });
 1152: 
 1153:         // Add legend with interactivity
 1154:         data.forEach((series, i) => {
 1155:             const legend = legendArea.append("g")
 1156:                 .attr("transform", `translate(0, \${i * 20})`)
 1157:                 .attr("class", "legend")
 1158:                 .on("click", () => {
 1159:                     const isVisible = d3.selectAll(\`path.line-\${i}\`).style("opacity") === "1";
 1160: 
 1161:                     // Toggle visibility
 1162:                     d3.selectAll(\`path.line-\${i}\`).style("opacity", isVisible ? 0 : 1);
 1163:                     d3.selectAll(\`circle.series-\${i}\`).style("opacity", isVisible ? 0 : 1);
 1164: 
 1165:                     // Dim legend if series is hidden
 1166:                     legend.select("text").style("opacity", isVisible ? 0.5 : 1);
 1167:                 });
 1168: 
 1169:             legend.append("rect")
 1170:                 .attr("width", 12)
 1171:                 .attr("height", 12)
 1172:                 .attr("fill", color(i));
 1173: 
 1174:             legend.append("text")
 1175:                 .attr("x", 20)
 1176:                 .attr("y", 10)
 1177:                 .text(series.name)
 1178:                 .style("alignment-baseline", "middle");
 1179:         });
 1180:     </script>
 1181: </body>
 1182: </html>
 1183: HTML
 1184: 
 1185:     return $html;
 1186: }
